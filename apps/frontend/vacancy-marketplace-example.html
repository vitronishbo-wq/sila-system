PASSO 9 - Consulte vagas disponíveis em tempo real

Buscar Vagas

Pesquise vagas disponíveis por instituição, série ou ano letivo. Os dados vêm da base de dados em tempo real.

Exemplos de Requisição

Endpoint HTTP GET: /api/v1/educacao/vacancies/

GET /api/v1/educacao/vacancies/
Listar todas as vagas
GET /api/v1/educacao/vacancies/?institution_id=school-123
Listar vagas de uma instituição específica
GET /api/v1/educacao/vacancies/?class_name=9
Listar vagas de uma série específica
GET /api/v1/educacao/vacancies/?institution_id=school-123&class_name=9
Pesquisa combinada (instituição + série)

Exemplo de Código JavaScript

// Opção 1: Usando Fetch API (moderno) async function fetchVacancies(filters = {}) { const params = new URLSearchParams(); if (filters.institution_id) params.append('institution_id', filters.institution_id); if (filters.class_name) params.append('class_name', filters.class_name); if (filters.academic_year) params.append('academic_year', filters.academic_year); try { const response = await fetch( `/api/v1/educacao/vacancies/?${params}`, { method: 'GET', headers: { 'Content-Type': 'application/json', // 'Authorization': 'Bearer YOUR_TOKEN' // Se necessário } } ); if (!response.ok) { throw new Error(`HTTP ${response.status}`); } const vacancies = await response.json(); return vacancies; } catch (error) { console.error('Erro ao buscar vagas:', error); return []; } } // Opção 2: Usando XMLHttpRequest (compatível com navegadores antigos) function fetchVacanciesLegacy(filters = {}) { const xhr = new XMLHttpRequest(); const params = new URLSearchParams(); if (filters.institution_id) params.append('institution_id', filters.institution_id); if (filters.class_name) params.append('class_name', filters.class_name); xhr.open('GET', `/api/v1/educacao/vacancies/?${params}`, true); xhr.setRequestHeader('Content-Type', 'application/json'); xhr.onload = function() { if (xhr.status === 200) { const vacancies = JSON.parse(xhr.responseText); console.log('Vagas disponíveis:', vacancies); } }; xhr.onerror = function() { console.error('Erro na requisição'); }; xhr.send(); } // Exemplo de uso: const vacancies = await fetchVacancies({ institution_id: 'school-123', class_name: '9' }); console.log(`Encontradas ${vacancies.length} vagas`); vacancies.forEach(v => { console.log(`${v.grade} em ${v.institution}: ${v.available_slots} vagas`) });

Exemplos em Outras Linguagens

Python (requests)

import requests # Buscar vagas de uma instituição específica response = requests.get( 'http://localhost:8000/api/v1/educacao/vacancies/', params={ 'institution_id': 'school-123', 'class_name': '9' } ) if response.status_code == 200: vacancies = response.json() for v in vacancies: print(f"{v['grade']} - {v['available_slots']} vagas disponíveis") else: print(f"Erro: {response.status_code}")

React Component

import React, { useState, useEffect } from 'react'; export function VacanciesList() { const [vacancies, setVacancies] = useState([]); const [loading, setLoading] = useState(false); useEffect(() => { setLoading(true); fetch('/api/v1/educacao/vacancies/') .then(res => res.json()) .then(data => setVacancies(data)) .finally(() => setLoading(false)); }, []); if (loading) return <p>Carregando...</p>; return ( <table> <thead> <tr> <th>Instituição</th> <th>Série</th> <th>Vagas Disponíveis</th> </tr> </thead> <tbody> {vacancies.map((v, i) => ( <tr key={i}> <td>{v.institution}</td> <td>{v.grade}</td> <td>{v.available_slots}</td> </tr> ))} </tbody> </table> ); }

Formato de Resposta

A API retorna um array de objetos JSON com a seguinte estrutura:

[ { "institution": "uuid-ou-id-da-escola", "grade": "9", "available_slots": 5 }, { "institution": "uuid-ou-id-da-escola-2", "grade": "5A", "available_slots": 0 } ] // Campos: // - institution (string): Identificador único da instituição // - grade (string): Série/classe (ex: "9", "5A", "10B") // - available_slots (integer): Quantidade de vagas abertas
